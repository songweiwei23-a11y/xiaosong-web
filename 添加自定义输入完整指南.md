  // 优化的多选+自定义组件
  const MultiSelectWithCustom = ({ 
    field, 
    label, 
    options, 
    placeholder,
    columns = 3
  }: { 
    field: string
    label: string
    options: string[]
    placeholder: string
    columns?: number
  }) => {
    const [customInput, setCustomInput] = useState('')
    const selectedValues = (formData[field as keyof typeof formData] as string[]) || []

    const toggleOption = (value: string) => {
      setFormData(prev => {
        const currentArray = prev[field as keyof typeof prev] as string[]
        return {
          ...prev,
          [field]: currentArray.includes(value)
            ? currentArray.filter(v => v !== value)
            : [...currentArray, value]
        }
      })
    }

    const addCustom = () => {
      if (customInput.trim()) {
        const newValues = customInput.split(/[,，、\s]+/).map(v => v.trim()).filter(v => v)
        setFormData(prev => {
          const currentArray = prev[field as keyof typeof prev] as string[]
          const uniqueValues = [...new Set([...currentArray, ...newValues])]
          return { ...prev, [field]: uniqueValues }
        })
        setCustomInput('')
      }
    }

    const customTags = selectedValues.filter(v => !options.includes(v))

    return (
      <div className=\"space-y-3\">
        <label className=\"block text-sm font-medium text-gray-900\">{label}</label>
        
        {/* 预设选项 */}
        <div className={\grid grid-cols-\ gap-2\}>
          {options.map((option) => (
            <button
              key={option}
              type=\"button\"
              onClick={() => toggleOption(option)}
              className={\px-4 py-2.5 rounded-lg border-2 text-sm font-medium transition-all \\}
            >
              {option}
            </button>
          ))}
        </div>

        {/* 自定义输入 */}
        <div className=\"bg-gradient-to-r from-purple-50 to-blue-50 p-4 rounded-lg border-2 border-dashed border-purple-200\">
          <div className=\"flex gap-2\">
            <input
              type=\"text\"
              value={customInput}
              onChange={(e) => setCustomInput(e.target.value)}
              onKeyPress={(e) => {
                if (e.key === 'Enter') {
                  e.preventDefault()
                  addCustom()
                }
              }}
              placeholder={placeholder}
              className=\"flex-1 px-4 py-2.5 border border-purple-300 rounded-lg focus:ring-2 focus:ring-purple-500 bg-white text-sm\"
            />
            <button
              type=\"button\"
              onClick={addCustom}
              className=\"px-6 py-2.5 bg-gradient-to-r from-purple-600 to-blue-600 text-white rounded-lg hover:from-purple-700 hover:to-blue-700 font-medium text-sm shadow-sm\"
            >
              ✚ 添加
            </button>
          </div>
          <p className=\"text-xs text-purple-600 mt-2\">💡 可输入多个，用逗号、顿号或空格分隔</p>
        </div>

        {/* 自定义标签 */}
        {customTags.length > 0 && (
          <div className=\"bg-green-50 border border-green-200 rounded-lg p-3\">
            <span className=\"text-sm font-medium text-green-800 block mb-2\">✓ 已添加自定义：</span>
            <div className=\"flex flex-wrap gap-2\">
              {customTags.map((tag, i) => (
                <span key={i} className=\"inline-flex items-center gap-1.5 px-3 py-1.5 bg-white border-2 border-green-300 text-green-700 rounded-full text-sm font-medium\">
                  {tag}
                  <button type=\"button\" onClick={() => toggleOption(tag)} className=\"hover:bg-green-200 rounded-full p-0.5\">×</button>
                </span>
              ))}
            </div>
          </div>
        )}
      </div>
    )
  }
