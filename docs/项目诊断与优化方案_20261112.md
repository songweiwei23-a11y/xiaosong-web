#### 🟡 问题 2：预留清理不统一 (P1)

**影响**: 数据表膨胀，额度可能被永久占用

```sql
-- 创建统一的预留清理函数
CREATE OR REPLACE FUNCTION cleanup_expired_reservations()
RETURNS void AS $$
BEGIN
  -- 1. 清理创作额度预留（2小时超时）
  UPDATE creation_requests 
  SET status = 'released' 
  WHERE status = 'pending' 
    AND expires_at <= now();
  
  -- 2. 清理联网搜索预留（2小时超时）
  UPDATE web_search_requests 
  SET status = 'released'
  WHERE status = 'reserved' 
    AND created_at < now() - interval '2 hours';
  
  RAISE NOTICE '清理完成：% 个创作预留，% 个搜索预留', 
    (SELECT COUNT(*) FROM creation_requests WHERE status = 'released'),
    (SELECT COUNT(*) FROM web_search_requests WHERE status = 'released');
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- 手动执行清理（推荐添加到后台定时任务）
SELECT cleanup_expired_reservations();
```

**建议**: 在 `/admin/settings` 添加"数据维护"功能，手动触发清理。

### 5.4 统一优化方案

**创建 API 工具库** `lib/api-utils.ts`:

```typescript
import { z } from 'zod';
import { NextResponse } from 'next/server';

// 1. 统一错误处理
export function apiError(message: string, status: number = 500) {
  return NextResponse.json({ error: message }, { status });
}

// 2. 统一验证
export async function validateBody<T>(
  request: Request,
  schema: z.ZodSchema<T>
): Promise<T | NextResponse> {
  try {
    const body = await request.json();
    return schema.parse(body);
  } catch (error) {
    if (error instanceof z.ZodError) {
      return apiError(
        '请求参数错误: ' + error.errors.map(e => e.message).join(', '),
        400
      );
    }
    return apiError('请求格式错误', 400);
  }
}

// 3. 统一用户验证
export async function requireUser(request: Request) {
  const user = await getUser(); // 你现有的函数
  if (!user) {
    throw new Response('Unauthorized', { status: 401 });
  }
  return user;
}

// 4. API 包装器（统一添加日志和错误处理）
export function apiHandler(
  handler: (request: Request) => Promise<Response>
) {
  return async (request: Request) => {
    const start = Date.now();
    const url = new URL(request.url);
    
    try {
      const response = await handler(request);
      const duration = Date.now() - start;
      
      console.log(`[API] ${request.method} ${url.pathname} ${response.status} ${duration}ms`);
      
      return response;
    } catch (error) {
      const duration = Date.now() - start;
      console.error(`[API] ${request.method} ${url.pathname} ERROR ${duration}ms`, error);
      
      if (error instanceof Response) {
        return error;
      }
      
      return apiError('服务器内部错误');
    }
  };
}
```

  const body = await validateBody(request, topicSchema);
  
  if (body instanceof NextResponse) return body; // 验证失败
  
  const result = await createTopic(user.id, body);
  return NextResponse.json(result);
