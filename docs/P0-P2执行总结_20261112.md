```typescript
// app/api/example/route.ts
import { apiHandler, validateBody, apiSuccess, apiError, schemas } from '@/lib/api-utils';

export const maxDuration = 30;

const createSchema = z.object({
  title: schemas.shortText,
  content: schemas.longText,
  userId: schemas.id,
});

export const POST = apiHandler(async (request) => {
  // 验证输入
  const { data, error } = await validateBody(request, createSchema);
  if (error) return error;

  try {
    // 业务逻辑
    const result = await createItem(data);
    return apiSuccess(result, 201);
  } catch (err) {

    return apiError('创建失败，请稍后重试');
  }
});
```
