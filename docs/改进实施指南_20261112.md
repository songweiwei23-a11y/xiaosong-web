1. **创建带时间预期的生成指示器**：
```typescript
// components/ui/generating-indicator.tsx
'use client';

import { useEffect, useState } from 'react';
import { LoadingRings } from './loading-rings';

interface GeneratingIndicatorProps {
  taskName?: string;
  estimatedSeconds?: number;
  onCancel?: () => void;
}

export function GeneratingIndicator({ 
  taskName = 'AI 生成',
  estimatedSeconds = 90,
  onCancel 
}: GeneratingIndicatorProps) {
  const [elapsed, setElapsed] = useState(0);

  useEffect(() => {
    const timer = setInterval(() => {
      setElapsed(s => s + 1);
    }, 1000);
    return () => clearInterval(timer);
  }, []);

  return (
    <div className="flex flex-col items-center justify-center p-8">
      <LoadingRings size={64} />
      <p className="mt-4 text-base font-medium">{taskName}中...</p>
      <p className="mt-1 text-sm text-muted-foreground">
        预计需要 {Math.ceil(estimatedSeconds / 60)} 分钟
      </p>
      
      {elapsed > 30 && (
        <p className="mt-2 text-sm text-amber-600">
          已等待 {elapsed} 秒，马上就好
        </p>
      )}
      
      {elapsed > estimatedSeconds * 1.5 && (
        <p className="mt-2 text-sm text-red-600">
          生成时间较长，可能网络或服务繁忙
        </p>
      )}
      
      {onCancel && elapsed > 10 && (
        <button 
          onClick={onCancel}
          className="mt-4 px-4 py-2 text-sm text-muted-foreground hover:text-foreground border rounded"
        >
          取消生成
        </button>
      )}
    </div>
  );
}
```

2. **在各页面中使用**：
```typescript
// app/dashboard/script/page.tsx

  <GeneratingIndicator 
    taskName="AI 正在生成脚本"
    estimatedSeconds={120}
    onCancel={handleCancel}
  />
)}
```

3. **实现取消功能**：

1. **创建统一的 API 客户端**：
```typescript
// lib/api-client.ts
import { notify } from '@/components/ui/feedback';

export class ApiError extends Error {
  constructor(
    message: string,
    public status: number,
    public code?: string
  ) {
    super(message);
    this.name = 'ApiError';
  }
}

export async function apiCall<T>(
  url: string,
  options?: RequestInit
): Promise<T> {
  try {
    const response = await fetch(url, {
      ...options,
      headers: {
        'Content-Type': 'application/json',
        ...options?.headers,
      },
    });

    if (!response.ok) {
      const error = await response.json().catch(() => ({}));
      throw new ApiError(
        error.error || `请求失败 (${response.status})`,
        response.status,
        error.code
      );
    }

    return response.json();
  } catch (error) {
    if (error instanceof ApiError) {
      throw error;
    }
    
    // 网络错误
    if (error instanceof TypeError && error.message.includes('fetch')) {
      throw new ApiError('网络连接失败，请检查网络后重试', 0);
    }
    
    throw new ApiError('未知错误，请重试', 500);
  }
}

export async function apiCallWithNotify<T>(
  url: string,
  options?: RequestInit,
  errorMessage?: string
): Promise<T | null> {
  try {
    return await apiCall<T>(url, options);
  } catch (error) {
    if (error instanceof ApiError) {
      notify(errorMessage || error.message, 'error');
    } else {
      notify('操作失败，请重试', 'error');
    }
    console.error('[API Error]', url, error);
    return null;
  }
}
```

2. **替换现有的 fetch 调用**：

// 改为

## 🎯 阶段2：系统稳定性（第2周）

### 任务2.1：为110个API添加超时设置

**实施步骤**：

1. **找出所有缺少超时的 API**：
```powershell
Get-ChildItem -Path "app/api" -Recurse -Filter "route.ts" | ForEach-Object {
  $content = Get-Content $_.FullName -Raw
  if ($content -notmatch "maxDuration") {
    Write-Host $_.FullName
  }
}
```

2. **批量添加超时**：
```typescript
// 在每个 route.ts 顶部添加
export const maxDuration = 30; // 根据实际调整

// 推荐值:
// 简单查询: 5-10 秒
// AI 生成: 30-60 秒
// 批量操作: 60-120 秒
// 上传文件: 120 秒
```

3. **创建自动化脚本**：
```powershell
# add-timeouts.ps1
$routes = Get-ChildItem -Path "app/api" -Recurse -Filter "route.ts"

foreach ($route in $routes) {
  $content = Get-Content $route.FullName -Raw
  
  if ($content -notmatch "maxDuration") {
    # 根据文件路径判断超时值
    $timeout = 30
    if ($route.FullName -match "dify|generate|script") { $timeout = 60 }
    if ($route.FullName -match "batch|bulk") { $timeout = 120 }
    
    # 在文件开头添加
    $newContent = "export const maxDuration = $timeout;`n`n" + $content
    Set-Content $route.FullName $newContent
    
    Write-Host "Added timeout to: $($route.FullName)"
  }
}
```

**预计时间**: 2-3 天  
**验收标准**:
- ✅ 所有 API 都有 maxDuration
- ✅ 超时值合理
- ✅ 测试通过
